function cheatPermission(){return gameMode!=='multi'||(onlineHost&&!!onlineRoom)}
function localKillAll(){
 let n=0;
 for(const e of enemies){
  if(!e.alive)continue;
  e.alive=false;e.hp=0;e.group.visible=false;e.respawn=37;
  setEnemyBar(e);n++;state.enemyDefeats[e.type]=(state.enemyDefeats[e.type]||0)+1;
  if(e.type==='preside')state.bossDefeated=true;
  if(e.type==='vice')state.viceDefeated=true;
  if(e.type==='bullo')state.bullyDefeated=true;
 }
 toast('💥 KILL ALL: '+n+' nemici KO!',2);safeSave();
}
function collectCandyAll(){
 let n=0;
 for(const c of [...pickups]){
  if(state.collected.includes(c.id))continue;
  n++;state.collected.push(c.id);scene.remove(c.mesh);
  const at=pickups.indexOf(c);if(at>=0)pickups.splice(at,1);
 }
 state.candy+=n;if(gameMode==='single')addCosmeticCandies(n*5);
 safeSave();toast('🍬 +'+n+' CARAMELLE!',2);
}

}

function applyCheat(key){
 if(!cheatPermission()){toast('🔒 Solo admin in multiplayer',2);return}
 if(!CHEAT_CATALOG.some(c=>c[0]===key))return;
 if(gameMode==='multi'&&!netReady){toast('⏳ Avvia la gara prima di usare i trucchi!',2);return}
 if(CHEAT_ACTIONS.has(key)){
  if(gameMode==='multi')sendNet({type:'hostCheat',action:key});
  else if(key==='killall')localKillAll();
  else if(key==='collectAll')collectCandyAll();
  else if(key==='fullHeal'){state.hp=100;toast('❤️ Salute al massimo!',2)}
  else if(key==='teleportHome')localTeleportHome();
 }else{
  cheats[key]=!cheats[key];
  if(gameMode==='multi'){
   sendNet({type:'hostCheat',action:key,enabled:cheats[key]});
   if(key==='freeze')roomFreeze=cheats[key];
  }
  if(key==='fly'){state.vy=0;state.airJumps=0}
  toast((cheats[key]?'✅ ':'❎ ')+key.toUpperCase(),1);
 }
 cheatMenu();
}
function updateCheatAimbot(){
 if(!cheats.aimbot||!cheatPermission())return;
 let best=null,min=70;
 if(gameMode==='multi'){
  if(!netReady)return;
  for(const r of remotePlayers.values()){
   const d=Math.hypot(state.x-r.group.position.x,state.z-r.group.position.z);
   if(d<min){min=d;best={x:r.group.position.x,y:r.group.position.y+1.4,z:r.group.position.z}}
  }
 }else for(const e of enemies){
   if(!e.alive||e.state==='underground')continue;
   const d=Math.hypot(state.x-e.group.position.x,state.z-e.group.position.z);
   if(d<min){min=d;best={x:e.group.position.x,y:e.group.position.y+1.5,z:e.group.position.z}}
 }
 if(!best)return;
 const dx=best.x-state.x,dy=best.y-state.y-1.43,dz=best.z-state.z;
 state.yaw=-Math.atan2(dx,-dz);
 state.pitch=clamp(Math.atan2(dy,Math.hypot(dx,dz)),-.43,.83);
 player.rotation.y=state.yaw;
}

  sun.color.setHSL((visualTime*.17)%1,.60,.73);
 }else sun.color.set('#ffffff');

 }else $('cheatRadar').style.display='none';
